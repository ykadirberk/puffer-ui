cmake_minimum_required(VERSION 3.21)
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
endif()
project(pufferui_consumer LANGUAGES CXX)
if(PUFFERUI_SOURCE)
  set(PUFFERUI_WITH_SDL3 OFF CACHE BOOL "" FORCE)
  add_subdirectory("${PUFFERUI_SOURCE}" pufferui)
  if(TARGET pui_demo OR TARGET pui_core_tests)
    message(FATAL_ERROR "Library consumption unexpectedly adds examples/tests")
  endif()
else()
  find_package(pufferui CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
if(CONSUMER_SDL)
  target_link_libraries(consumer PRIVATE pufferui::sdl3)
else()
  target_link_libraries(consumer PRIVATE pufferui::pufferui)
endif()
# Deliberately no CXX_STANDARD: the linked target must propagate C++20.
